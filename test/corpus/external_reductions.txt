ac_fpublic ac_EnableStuntBonusForAll(enable)
{
    ac_StuntBonus = !!enable;
#if defined ITERATOR
    AnyIterator(new ac_i : Player) ACInfo[ac_i][acStuntBonus] = ac_StuntBonus;
#else
    #if defined DYNAMIC_LIMIT
        for (new ac_i = GetPlayerPoolSize(); ac_i >= 0; --ac_i)
    #else
        for (new ac_i = MAX_PLAYERS - 1; ac_i >= 0; --ac_i)
    #endif
    {
        if (!IsPlayerConnected(ac_i)) continue;
        ACInfo[ac_i][acStuntBonus] = ac_StuntBonus;
    }
#endif
    EnableStuntBonusForAll(!!enable);
    return 1;
}

stock Visit(playerid) {
#if defined ITERATOR
    AnyIterator(new item : Players)
    {
#else
    #if defined DYNAMIC_LIMIT
        for (new item = GetLastPlayer(); item >= 0; item--)
    #else
        for (new item = MAX_PLAYERS - 1; item >= 0; item--)
    #endif
    {
#if USE_NPCS
        if (item == playerid || IsNpc(item)) continue;
#else
        if (item == playerid) continue;
#endif
#endif
        VisitPlayer(item);
    }
}
